ef findBestNumericSplit(self, dSet, feature_name, target_column):
    """
    Finds the best split point for a numerical feature.

    Parameters:
        dSet - The dataset.
        feature_name - The name of the numerical feature.
        target_column - The name of the target variable.

    Returns:
        The best split value.
    """
    # Sort the unique values of the feature
    unique_values = np.sort(dSet[feature_name].unique())
    best_info_gain = -1
    best_split_value = None

    # Iterate through potential split points (midpoints between sorted values)
    for i in range(len(unique_values) - 1):
        split_value = (unique_values[i] + unique_values[i+1]) / 2

        # Create temporary subsets
        subset_left = dSet[dSet[feature_name] <= split_value]
        subset_right = dSet[dSet[feature_name] > split_value]

        # Calculate information gain for this split
        info_gain = self.informationGain(dSet, [subset_left, subset_right], feature_name, is_numeric=True)

        if info_gain > best_info_gain:
            best_info_gain = info_gain
            best_split_value = split_value

    return best_split_value
